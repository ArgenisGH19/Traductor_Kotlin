package com.ejemplo;

class MiClase {
    
    fun sumar(a: int, b: int): int {
        return a + b;
    }
    
    fun esMayor(valor: int, limite: int): boolean {
        return valor > limite;
    }

    fun proceso() {
        var iteracion: int = 0;
        var activo: boolean = true;
        
        // Ciclo con condicion booleana correcta
        while (activo) {
            iteracion = iteracion + 1;
            
            // continue DENTRO de un ciclo (Valido)
            if (iteracion == 5) {
                escribir("Llegamos a la mitad");
                continue; 
            }
            
            // break DENTRO de un ciclo (Valido)
            if (iteracion >= 10) {
                break; 
            }
        }
        
        // Metodo llamado con los argumentos exactos (2 ints)
        var resultado: int = sumar(10, 20); 
        
        // Condicion booleana usando el retorno de un metodo
        if (esMayor(resultado, 15)) { 
            escribir("El resultado es mayor");
        }
    }
}
