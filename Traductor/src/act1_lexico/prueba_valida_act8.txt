package com.ejemplo;

class MiClase {
    
    fun iniciar() {
        escribir("Inicio del programa");
    }

    fun mostrar(valor: int) {
        escribir(valor);
    }

    fun sumar(a: int, b: int): int {
        return a + b;
    }

    fun obtenerValor(): int {
        return 100;
    }

    fun proceso() {
        iniciar();
        mostrar(10);
        
        // ¡Ahora sí las declaramos con var y su tipo!
        var resultado: int = sumar(5, 10);
        var valorObtenido: int = obtenerValor(); // Le cambié el nombre para no confundir con el 'valor' del otro método
        
        // Declaramos 'a' y 'b' locales para poder usarlas
        var a: int = 20;
        var b: int = 30;
        var total: int = sumar(a, b) + obtenerValor();
        
        escribir(resultado);
        escribir(valorObtenido);
    }
}
